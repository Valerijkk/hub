1. Создать backend/internal/auth/handler.go:
```go
package auth

import (
    "github.com/gofiber/fiber/v3"
)

type Handler struct {
    service *AuthService
}

func NewHandler(service *AuthService) *Handler {
    return &Handler{service: service}
}

func (h *Handler) RegisterRoutes(api fiber.Router, authMiddleware fiber.Handler) {
    auth := api.Group("/auth")

    // Публичные эндпоинты (без auth middleware)
    auth.Post("/magic-link", h.SendMagicLink)      // AUTH-01
    auth.Post("/magic-link/verify", h.VerifyMagicLink) // AUTH-01
    auth.Post("/telegram", h.LoginTelegram)          // AUTH-02
    auth.Post("/refresh", h.RefreshToken)

    // Защищённые эндпоинты
    authGroup := auth.Group("", authMiddleware)
    authGroup.Post("/logout", h.Logout)
    authGroup.Get("/me", h.GetMe)                    // AUTH-04
    authGroup.Put("/me", h.UpdateProfile)            // AUTH-04
    authGroup.Post("/me/onboarding", h.CompleteOnboarding) // AUTH-08
    authGroup.Get("/sessions", h.ListSessions)       // AUTH-06
    authGroup.Delete("/sessions/:sessionId", h.RevokeSession) // AUTH-06

    // Admin-only
    admin := api.Group("/admin", authMiddleware)
    admin.Get("/registration-mode", h.GetRegistrationMode) // AUTH-03
    admin.Put("/registration-mode", h.SetRegistrationMode) // AUTH-03
}
```
