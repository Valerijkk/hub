```go
package config

import (
    "os"
    "strconv"
    "time"
)

type Config struct {
    Server   ServerConfig
    Postgres PostgresConfig
    MongoDB  MongoDBConfig
    Redis    RedisConfig
    NATS     NATSConfig
    S3       S3Config
    JWT      JWTConfig
    Resend   ResendConfig
    Telegram TelegramConfig
}

type ServerConfig struct {
    Port         int
    ReadTimeout  time.Duration
    WriteTimeout time.Duration
    Immutable    bool // true in dev для отлова Fiber RequestCtx reuse (Pitfall 1)
}

type PostgresConfig struct {
    URL string
}

type MongoDBConfig struct {
    URL      string
    Database string
}

type RedisConfig struct {
    URL string
}

type NATSConfig struct {
    URL string
}

type S3Config struct {
    Endpoint  string
    AccessKey string
    SecretKey string
    Bucket    string
    Region    string
    UseSSL    bool
}

type JWTConfig struct {
    Secret          string
    AccessTokenTTL  time.Duration
    RefreshTokenTTL time.Duration
}

type ResendConfig struct {
    APIKey   string
    FromEmail string
}

type TelegramConfig struct {
    BotToken string
}

func Load() *Config {
    return &Config{
        Server: ServerConfig{
            Port:         getEnvInt("SERVER_PORT", 3000),
            ReadTimeout:  10 * time.Second,
            WriteTimeout: 10 * time.Second,
            Immutable:    getEnvBool("FIBER_IMMUTABLE", true),
        },
        Postgres: PostgresConfig{
            URL: getEnv("DATABASE_URL", "postgresql://hub:hub@localhost:5432/hub?sslmode=disable"),
        },
        MongoDB: MongoDBConfig{
            URL:      getEnv("MONGODB_URL", "mongodb://hub:hub@localhost:27017"),
            Database: getEnv("MONGODB_DATABASE", "hub"),
        },
        Redis: RedisConfig{
            URL: getEnv("REDIS_URL", "redis://localhost:6379/0"),
        },
        NATS: NATSConfig{
            URL: getEnv("NATS_URL", "nats://localhost:4222"),
        },
        S3: S3Config{
            Endpoint:  getEnv("S3_ENDPOINT", "http://localhost:8333"),
            AccessKey: getEnv("S3_ACCESS_KEY", "hub"),
            SecretKey: getEnv("S3_SECRET_KEY", "hubsecret"),
            Bucket:    getEnv("S3_BUCKET", "hub"),
            Region:    getEnv("S3_REGION", "us-east-1"),
            UseSSL:    getEnvBool("S3_USE_SSL", false),
        },
        JWT: JWTConfig{
            Secret:          getEnv("JWT_SECRET", "dev-secret-change-in-production"),
            AccessTokenTTL:  15 * time.Minute,
            RefreshTokenTTL: 30 * 24 * time.Hour,
        },
        Resend: ResendConfig{
            APIKey:    getEnv("RESEND_API_KEY", ""),
            FromEmail: getEnv("RESEND_FROM_EMAIL", "noreply@hub.local"),
        },
        Telegram: TelegramConfig{
            BotToken: getEnv("TELEGRAM_BOT_TOKEN", ""),
        },
    }
}

func getEnv(key, fallback string) string {
    if v := os.Getenv(key); v != "" {
        return v
    }
    return fallback
}

func getEnvInt(key string, fallback int) int {
    if v := os.Getenv(key); v != "" {
        if i, err := strconv.Atoi(v); err == nil {
            return i
        }
    }
    return fallback
}

func getEnvBool(key string, fallback bool) bool {
    if v := os.Getenv(key); v != "" {
        if b, err := strconv.ParseBool(v); err == nil {
            return b
        }
    }
    return fallback
}
```

5. Создать backend/cmd/server/main.go:
```go
package main

import (
    "fmt"
    "log/slog"
    "os"
    "os/signal"
    "syscall"

    "github.com/gofiber/fiber/v3"
    "github.com/valerii/hub/backend/internal/platform/config"
    "github.com/valerii/hub/backend/internal/platform/middleware"
)

func main() {
    logger := slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{Level: slog.LevelInfo}))
    slog.SetDefault(logger)

    cfg := config.Load()

    app := fiber.New(fiber.Config{
        ReadTimeout:  cfg.Server.ReadTimeout,
        WriteTimeout: cfg.Server.WriteTimeout,
        Immutable:    cfg.Server.Immutable,
        ErrorHandler: middleware.ErrorHandler,
    })

    app.Use(middleware.RequestID())
    app.Use(middleware.Logger())
    app.Use(middleware.CORS())

    api := app.Group("/api")
    api.Get("/health", func(c fiber.Ctx) error {
        return c.JSON(fiber.Map{"status": "ok"})
    })

    go func() {
        addr := fmt.Sprintf(":%d", cfg.Server.Port)
        slog.Info("starting server", "addr", addr)
        if err := app.Listen(addr); err != nil {
            slog.Error("server failed", "error", err)
            os.Exit(1)
        }
    }()

    quit := make(chan os.Signal, 1)
    signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
    <-quit

    slog.Info("shutting down server")
    if err := app.Shutdown(); err != nil {
        slog.Error("server shutdown error", "error", err)
    }
}
```

6. Создать backend/internal/platform/middleware/ файлы:

logging.go -- slog middleware логирующий method, path, status, duration, request_id.

cors.go -- CORS middleware: AllowOrigins "http://localhost:5173" (Vite dev), AllowHeaders "Content-Type, Authorization", AllowMethods "GET, POST, PUT, PATCH, DELETE, OPTIONS".

ratelimit.go -- in-memory rate limiter (sync.Map + sliding window). Не зависит от Redis на старте. Конфиги: auth 3/10min, api 100/sec, files 10/min, messages 60/min. На данном этапе только auth и api лимиты, остальные добавятся позже.

middleware.go -- ErrorHandler функция для Fiber: проверяет тип ошибки (AppError vs fiber.Error vs generic error), возвращает JSON с code, message, request_id.
