type Client struct {

}

func NewClient(ctx context.Context, url string) (*Client, error) {
    nc, err := nats.Connect(url,
        nats.Name("hub-backend"),
        nats.MaxReconnects(-1),
    )

    }

    js, err := jetstream.New(nc)

    return c.Conn.Publish(subject, data)
}

func (c *Client) Subscribe(subject string, handler nats.MsgHandler) (*nats.Subscription, error) {
    return c.Conn.Subscribe(subject, handler)
}

var DefaultRolePermissions = map[string]Permission{
    "owner":     ^Permission(0), // все разрешения (0xFFFFFFFFFFFFFFFF)
    "developer": PermProjectView | PermTaskCreate | PermTaskEdit | PermTaskAssign | PermTaskComment | PermFileUpload | PermFileDelete | PermChannelCreate | PermMessageEdit | PermTestCaseCreate | PermTestCaseEdit | PermTestRunCreate | PermTestRunExecute | PermInviteCreate,

func (p Permission) Has(perm Permission) bool {
    return p&perm == perm

func (p Permission) Add(perm Permission) Permission {
    return p | perm

func (p Permission) Remove(perm Permission) Permission {

// PublishInTx записывает событие в outbox таблицу внутри существующей транзакции.
// Используется когда нужно гарантировать доставку события вместе с основной записью.
func (p *Publisher) PublishInTx(ctx context.Context, tx *ent.Tx, eventType string, payload any) error {
    data, err := json.Marshal(payload)
    if err != nil {
        return err
    }
    _, err = tx.OutboxEvent.Create().
        SetEventType(eventType).
        SetPayload(data).
        Save(ctx)
    return err
}

package outbox

import (
    "context"
    "log/slog"
    "time"

    "github.com/valerii/hub/backend/ent"
    "github.com/valerii/hub/backend/ent/outboxevent"
    natsclient "github.com/valerii/hub/backend/internal/platform/nats"
)

type Worker struct {
    client *ent.Client
    nats   *natsclient.Client
}

func NewWorker(client *ent.Client, nats *natsclient.Client) *Worker {
    return &Worker{client: client, nats: nats}
}

// ProcessLoop поллит outbox таблицу каждые 100ms и публикует pending события в NATS.
// При ошибке публикации инкрементирует attempts и записывает last_error.
// События с attempts > 10 помечаются как failed.
func (w *Worker) ProcessLoop(ctx context.Context) {
    ticker := time.NewTicker(100 * time.Millisecond)
    defer ticker.Stop()

    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            w.processBatch(ctx)
        }
    }
}

func (w *Worker) processBatch(ctx context.Context) {
    events, err := w.client.OutboxEvent.Query().
        Where(
            outboxevent.StatusEQ(outboxevent.StatusPending),
            outboxevent.AttemptsLT(10),
        ).
        Order(ent.Asc(outboxevent.FieldCreatedAt)).
        Limit(50).
        All(ctx)
    if err != nil {
        slog.Error("outbox: failed to query events", "error", err)
        return
    }

    for _, evt := range events {
        if err := w.nats.Publish(evt.EventType, evt.Payload); err != nil {

                SetLastError(err.Error()).
                SaveX(ctx)
            continue
        }
        now := time.Now()
        evt.Update().
            SetStatus(outboxevent.StatusPublished).
            SetPublishedAt(now).
            SaveX(ctx)
