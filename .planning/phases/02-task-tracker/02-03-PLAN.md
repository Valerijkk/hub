/** Board filter state for Kanban view. */
interface BoardFilters {
  status?: string[];
  priority?: string[];
  assignee?: string;
  type?: string;
  search?: string;
  myTasks?: boolean;
}

/** Board store manages active board filters and drag state. */
interface BoardState {
  /** Currently active board ID. */
  activeBoardId: string | null;
  /** Current filter configuration. */
  filters: BoardFilters;
  /** Whether a card is currently being dragged. */
  isDragging: boolean;
  /** ID of the issue currently being dragged. */
  draggingIssueId: string | null;

  setActiveBoard: (boardId: string) => void;
  setFilters: (filters: Partial<BoardFilters>) => void;
  resetFilters: () => void;
  setDragging: (isDragging: boolean, issueId?: string | null) => void;
}

const defaultFilters: BoardFilters = {};

export const useBoardStore = create<BoardState>()((set) => ({
  activeBoardId: null,
  filters: defaultFilters,
  isDragging: false,
  draggingIssueId: null,

  setActiveBoard: (boardId) => set({ activeBoardId: boardId }),
  setFilters: (filters) =>
    set((state) => ({ filters: { ...state.filters, ...filters } })),
  resetFilters: () => set({ filters: defaultFilters }),
  setDragging: (isDragging, issueId = null) =>
    set({ isDragging, draggingIssueId: issueId }),
}));

import { create } from "zustand";
import { persist } from "zustand/middleware";

/** Timer store manages the active time tracking timer. */
interface TimerState {
  /** Whether the timer is running. */
  isRunning: boolean;
  /** Issue key the timer is tracking (e.g., "HUB-42"). */
  activeIssueKey: string | null;
  /** Project ID for the active timer. */
  activeProjectId: string | null;
  /** Timestamp when the timer was started (ISO string). */
  startedAt: string | null;
  /** Elapsed seconds (updated by interval). */
  elapsedSeconds: number;

  startTimer: (projectId: string, issueKey: string) => void;
  stopTimer: () => void;
  tick: () => void;
}

export const useTimerStore = create<TimerState>()(
  persist(
    (set, get) => ({
      isRunning: false,
      activeIssueKey: null,
      activeProjectId: null,
      startedAt: null,
      elapsedSeconds: 0,

      startTimer: (projectId, issueKey) =>
        set({
          isRunning: true,
          activeIssueKey: issueKey,
          activeProjectId: projectId,
          startedAt: new Date().toISOString(),
          elapsedSeconds: 0,
        }),

      stopTimer: () =>
        set({
          isRunning: false,
          activeIssueKey: null,
          activeProjectId: null,
          startedAt: null,
          elapsedSeconds: 0,
        }),

      tick: () => {
        const { startedAt, isRunning } = get();
        if (!isRunning || !startedAt) return;
        const elapsed = Math.floor(
          (Date.now() - new Date(startedAt).getTime()) / 1000
        );
        set({ elapsedSeconds: elapsed });
      },
    }),
    { name: "hub-timer" }
  )
);
```

import { useCallback, useMemo } from "react";
import { Plate, PlateContent, type PlateEditor as PlateEditorType, type Value } from "@udecode/plate/react";
import { cn } from "@/lib/utils";

/** Props for the PlateEditor component. */
interface PlateEditorProps {
  /** Initial Plate JSON value. */
  value?: Value;
  /** Callback when content changes. */
  onChange?: (value: Value) => void;
  /** Placeholder text when editor is empty. */
  placeholder?: string;
  /** Whether the editor is read-only. */
  readOnly?: boolean;
  /** Additional CSS classes for the editor container. */
  className?: string;
}

/**
 * PlateEditor wraps the Plate rich-text editor with Hub-specific configuration.
 * Supports basic formatting (bold, italic, code, lists) and @mentions.
 * Stores content as Plate JSON for backend persistence.
 */
export function PlateEditor({
  value,
  onChange,
  placeholder = "Введите текст...",
  readOnly = false,
  className,
}: PlateEditorProps) {
  const handleChange = useCallback(
    ({ value: newValue }: { value: Value }) => {
      onChange?.(newValue);
    },
    [onChange]
  );

  const initialValue = useMemo(
    () => value || [{ type: "p", children: [{ text: "" }] }],
    [value]
  );

  return (
    <div className={cn("rounded-md border", className)}>
      <Plate
        value={initialValue}
        onChange={handleChange}
        readOnly={readOnly}
      >
        <PlateContent
          placeholder={placeholder}
          className={cn(
            "min-h-[120px] p-3 text-sm focus:outline-none",
            readOnly && "cursor-default"
          )}
        />
      </Plate>
    </div>
  );
}
```
