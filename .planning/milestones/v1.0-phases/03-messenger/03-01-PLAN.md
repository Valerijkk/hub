  GET    /:channelId/members  -> ListMembers
messages := channels.Group("/:channelId/messages")
  POST   /                    -> SendMessage
  GET    /                    -> ListMessages
  PUT    /:messageId          -> EditMessage (PermMessageEdit)
  DELETE /:messageId          -> DeleteMessage
  POST   /:messageId/reactions -> ToggleReaction
  POST   /:messageId/pin      -> TogglePin
  POST   /:messageId/forward  -> ForwardMessage
  GET    /:messageId/replies   -> ListThreadReplies
msg.GET  /search               -> SearchMessages
msg.GET  /bookmarks            -> ListBookmarks
msg.POST /bookmarks            -> AddBookmark
msg.DELETE /bookmarks/:bookmarkId -> RemoveBookmark
msg.POST /dm                   -> CreateOrGetDM
